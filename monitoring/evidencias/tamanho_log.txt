# script:
import json, statistics
linhas = open("logs/inferencia.jsonl", "rb").read().splitlines()
grupos = {"pedido": [], "resposta ok": [], "resposta 400 sem entrada": [], "resposta 400 com entrada": []}
for l in linhas:
    e = json.loads(l); b = len(l) + 1
    if e["evento"] == "pedido": grupos["pedido"].append(b)
    elif e["status"] == "ok": grupos["resposta ok"].append(b)
    elif e["entrada"] is None: grupos["resposta 400 sem entrada"].append(b)
    else: grupos["resposta 400 com entrada"].append(b)
print(f"linhas no logs/inferencia.jsonl: {len(linhas)}")
for k, v in grupos.items():
    if v: print(f"{k:26s} n={len(v):5d}  média {statistics.mean(v):7.1f} bytes  (mín {min(v)}, máx {max(v)})")
par = statistics.mean(grupos["pedido"]) + statistics.mean(grupos["resposta ok"])
print(f"pedido + resposta ok = {par:.0f} bytes por pedido -> {10*1024*1024/par:.0f} pedidos por arquivo de 10 MB, {6*10*1024*1024/par:.0f} em 6 arquivos (atual + 5 backups)")
